# AI training and content-use policy for MeStory (https://mestory-ai.com)
# Format inspired by https://spawning.ai/ai.txt and similar emerging conventions.
# Last updated: 2026-04-26

# General principle:
#   The marketing site (landing pages, FAQ, guides, marketplace listings, llms.txt,
#   llms-full.txt) is intended to be discoverable, indexable, and citable by AI
#   search engines and assistants. We WELCOME citations that link back to
#   https://mestory-ai.com.
#
#   User-generated content (the books themselves, author bios written by users)
#   belongs to the respective authors and should not be used for model training
#   or reproduced without permission.

User-Agent: *
Allow: /
Allow: /marketplace
Allow: /about
Allow: /faq
Allow: /guides

Allow: /subscription
Allow: /privacy

Allow: /llms.txt
Allow: /llms-full.txt

# user-generated content (book text, photos uploaded by families).

Disallow: /book/*/preview
